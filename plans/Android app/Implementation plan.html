FileSystemManager.kt (key methods)

class FileSystemManager(private val context: Context) {
    private var dataFolder: File? = null
    
    fun setDataFolder(folder: File) {
        this.dataFolder = folder
    }
    
    fun openFile(relativePath: String): String {
        val file = File(dataFolder, relativePath)
        require(isPathSafe(file)) { "Path traversal attempt" }
        return file.readText()
    }
    
    private fun isPathSafe(file: File): Boolean {
        return file.canonicalPath.startsWith(dataFolder?.canonicalPath ?: "")
    }

Risks and Mitigations

Risk Mitigation
WebView performance on older devices Caching, load optimization
Android file system restrictions Using the Storage Access Framework
Compatibility with different Android versions Minimum API version 24, testing
Path traversal security Strict path validation, canonical path
Memory consumption by large notes Incremental loading, lazy loading
